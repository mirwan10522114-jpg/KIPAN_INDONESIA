     1→"use client";
     2→
     3→import { useState, useEffect, useMemo } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  Award,
     7→  Plus,
     8→  Search,
     9→  RefreshCw,
    10→  Edit,
    11→  Trash2,
    12→  X,
    13→  Save,
    14→  ChevronDown,
    15→  ChevronRight,
    16→  Users,
    17→  AlertCircle,
    18→} from "lucide-react";
    19→import { toast } from "sonner";
    20→
    21→const LEVELS = ["Nasional", "Provinsi", "Kabupaten"] as const;
    22→
    23→export default function JabatanPage() {
    24→  const [jabatanList, setJabatanList] = useState<any[]>([]);
    25→  const [loading, setLoading] = useState(true);
    26→  const [search, setSearch] = useState("");
    27→  const [levelFilter, setLevelFilter] = useState<string>("Semua");
    28→  const [expandedBidang, setExpandedBidang] = useState<Record<string, boolean>>({});
    29→  const [showFormDialog, setShowFormDialog] = useState(false);
    30→  const [editingItem, setEditingItem] = useState<any>(null);
    31→  const [showAddBidangDialog, setShowAddBidangDialog] = useState(false);
    32→
    33→  const fetchData = async () => {
    34→    setLoading(true);
    35→    try {
    36→      const res = await fetch("/api/jabatan", { cache: "no-store" });
    37→      const json = await res.json();
    38→      if (json.success) setJabatanList(json.data);
    39→    } catch (e) {
    40→      console.error(e);
    41→      toast.error("Gagal memuat data jabatan");
    42→    } finally {
    43→      setLoading(false);
    44→    }
    45→  };
    46→
    47→  useEffect(() => {
    48→    fetchData();
    49→  }, []);
    50→
    51→  // Filter
    52→  const filtered = useMemo(() => {
    53→    let result = jabatanList;
    54→    if (levelFilter !== "Semua") {
    55→      result = result.filter((j) => j.level === levelFilter);
    56→    }
    57→    if (search) {
    58→      const q = search.toLowerCase();
    59→      result = result.filter(
    60→        (j) => j.nama.toLowerCase().includes(q) || j.bidang.toLowerCase().includes(q)
    61→      );
    62→    }
    63→    return result;
    64→  }, [jabatanList, levelFilter, search]);
    65→
    66→  // Group by bidang
    67→  const grouped = useMemo(() => {
    68→    const g: Record<string, any[]> = {};
    69→    for (const j of filtered) {
    70→      if (!g[j.bidang]) g[j.bidang] = [];
    71→      g[j.bidang].push(j);
    72→    }
    73→    // Sort urutan in each bidang
    74→    for (const b of Object.keys(g)) {
    75→      g[b].sort((a, b2) => a.urutan - b2.urutan);
    76→    }
    77→    return g;
    78→  }, [filtered]);
    79→
    80→  const bidangNames = Object.keys(grouped).sort();
    81→
    82→  // Stats
    83→  const totalJabatan = filtered.length;
    84→  const totalBidang = bidangNames.length;
    85→  const totalPengurus = filtered.reduce((sum, j) => sum + (j._count?.pengurus || 0), 0);
    86→  const totalAktif = filtered.filter((j) => j.status === "Aktif").length;
    87→
    88→  const toggleBidang = (bidang: string) => {
    89→    setExpandedBidang((prev) => ({ ...prev, [bidang]: !prev[bidang] }));
    90→  };
    91→
    92→  const handleDelete = async (id: number, nama: string, bidang: string) => {
    93→    if (!confirm(`Yakin hapus jabatan "${nama}" dari bidang "${bidang}"?`)) return;
    94→    try {
    95→      const res = await fetch(`/api/jabatan/${id}`, { method: "DELETE" });
    96→      const json = await res.json();
    97→      if (json.success) {
    98→        toast.success(json.message);
    99→        fetchData();
   100→      } else {
   101→        toast.error(json.error);
   102→      }
   103→    } catch (e: any) {
   104→      toast.error(e.message);
   105→    }
   106→  };
   107→
   108→  const handleToggleStatus = async (item: any) => {
   109→    const newStatus = item.status === "Aktif" ? "Nonaktif" : "Aktif";
   110→    try {
   111→      const res = await fetch(`/api/jabatan/${item.id}`, {
   112→        method: "PATCH",
   113→        headers: { "Content-Type": "application/json" },
   114→        body: JSON.stringify({ status: newStatus }),
   115→      });
   116→      const json = await res.json();
   117→      if (json.success) {
   118→        toast.success(`Jabatan ${newStatus === "Aktif" ? "diaktifkan" : "dinonaktifkan"}`);
   119→        fetchData();
   120→      } else {
   121→        toast.error(json.error);
   122→      }
   123→    } catch (e: any) {
   124→      toast.error(e.message);
   125→    }
   126→  };
   127→
   128→  return (
   129→    <div className="space-y-6">
   130→      {/* Header */}
   131→      <div className="flex items-center justify-between">
   132→        <div>
   133→          <h1 className="text-2xl font-bold text-blue-950">Bidang & Jabatan</h1>
   134→          <p className="text-slate-500 text-sm mt-1">
   135→            Kelola struktur organisasi — Pengurus Harian, Divisi-divisi, dan jabatan di tiap level
   136→          </p>
   137→        </div>
   138→        <div className="flex gap-2">
   139→          <button
   140→            onClick={fetchData}
   141→            className="inline-flex items-center gap-2 px-3 py-2 text-sm font-medium text-slate-600 hover:text-blue-700 hover:bg-blue-50 rounded-lg"
   142→          >
   143→            <RefreshCw className={`w-4 h-4 ${loading ? "animate-spin" : ""}`} /> Refresh
   144→          </button>
   145→          <button
   146→            onClick={() => {
   147→              setEditingItem(null);
   148→              setShowFormDialog(true);
   149→            }}
   150→            className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 text-white text-sm font-semibold rounded-lg hover:bg-blue-700"
   151→          >
   152→            <Plus className="w-4 h-4" /> Tambah Jabatan
   153→          </button>
   154→          <button
   155→            onClick={() => setShowAddBidangDialog(true)}
   156→            className="inline-flex items-center gap-2 px-4 py-2 bg-violet-600 text-white text-sm font-semibold rounded-lg hover:bg-violet-700"
   157→          >
   158→            <Plus className="w-4 h-4" /> Tambah Bidang Baru
   159→          </button>
   160→        </div>
   161→      </div>
   162→
   163→      {/* Stat Cards */}
   164→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   165→        <StatCard label="Total Bidang" value={totalBidang} color="text-violet-600" bg="bg-violet-50" />
   166→        <StatCard label="Total Jabatan" value={totalJabatan} color="text-blue-600" bg="bg-blue-50" />
   167→        <StatCard label="Jabatan Aktif" value={totalAktif} color="text-emerald-600" bg="bg-emerald-50" />
   168→        <StatCard label="Dipakai Pengurus" value={totalPengurus} color="text-amber-600" bg="bg-amber-50" />
   169→      </div>
   170→
   171→      {/* Filter */}
   172→      <div className="bg-white p-4 rounded-xl border border-slate-200 flex flex-wrap gap-3">
   173→        <div className="relative flex-1 min-w-[200px]">
   174→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   175→          <input
   176→            type="text"
   177→            value={search}
   178→            onChange={(e) => setSearch(e.target.value)}
   179→            placeholder="Cari jabatan atau bidang..."
   180→            className="w-full pl-9 pr-4 py-2 text-sm bg-slate-50 border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   181→          />
   182→        </div>
   183→        <select
   184→          value={levelFilter}
   185→          onChange={(e) => setLevelFilter(e.target.value)}
   186→          className="px-3 py-2 text-sm bg-slate-50 border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   187→        >
   188→          <option value="Semua">Semua Level</option>
   189→          {LEVELS.map((l) => <option key={l} value={l}>{l}</option>)}
   190→        </select>
   191→      </div>
   192→
   193→      {/* Bidang List */}
   194→      <div className="space-y-3">
   195→        {loading ? (
   196→          <div className="bg-white rounded-2xl p-12 text-center">
   197→            <div className="w-8 h-8 border-2 border-blue-200 border-t-blue-600 rounded-full animate-spin mx-auto mb-3" />
   198→            <p className="text-sm text-slate-500">Memuat data...</p>
   199→          </div>
   200→        ) : bidangNames.length === 0 ? (
   201→          <div className="bg-white rounded-2xl p-12 text-center">
   202→            <Award className="w-12 h-12 text-slate-300 mx-auto mb-3" />
   203→            <p className="text-sm text-slate-500 mb-3">Belum ada bidang & jabatan</p>
   204→            <button
   205→              onClick={() => setShowAddBidangDialog(true)}
   206→              className="inline-flex items-center gap-2 px-4 py-2 bg-violet-600 text-white text-sm font-semibold rounded-lg hover:bg-violet-700"
   207→            >
   208→              <Plus className="w-4 h-4" /> Tambah Bidang Baru
   209→            </button>
   210→          </div>
   211→        ) : (
   212→          bidangNames.map((bidang) => {
   213→            const items = grouped[bidang];
   214→            const isExpanded = expandedBidang[bidang] !== false; // default expand
   215→            const pengurusCount = items.reduce((s, j) => s + (j._count?.pengurus || 0), 0);
   216→            return (
   217→              <div key={bidang} className="bg-white rounded-2xl border border-slate-200 overflow-hidden">
   218→                <button
   219→                  onClick={() => toggleBidang(bidang)}
   220→                  className="w-full flex items-center gap-3 p-4 hover:bg-slate-50 transition-colors"
   221→                >
   222→                  <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-violet-500 to-purple-500 flex items-center justify-center shrink-0">
   223→                    <Award className="w-5 h-5 text-white" />
   224→                  </div>
   225→                  <div className="flex-1 text-left min-w-0">
   226→                    <div className="text-sm font-bold text-blue-950 truncate">{bidang}</div>
   227→                    <div className="text-xs text-slate-500 mt-0.5">
   228→                      {items.length} jabatan • {pengurusCount} pengurus terdaftar
   229→                    </div>
   230→                  </div>
   231→                  {isExpanded ? (
   232→                    <ChevronDown className="w-5 h-5 text-slate-400" />
   233→                  ) : (
   234→                    <ChevronRight className="w-5 h-5 text-slate-400" />
   235→                  )}
   236→                </button>
   237→                <AnimatePresence>
   238→                  {isExpanded && (
   239→                    <motion.div
   240→                      initial={{ height: 0, opacity: 0 }}
   241→                      animate={{ height: "auto", opacity: 1 }}
   242→                      exit={{ height: 0, opacity: 0 }}
   243→                      className="overflow-hidden"
   244→                    >
   245→                      <div className="border-t border-slate-100 divide-y divide-slate-50">
   246→                        {items.map((j) => (
   247→                          <div key={j.id} className="flex items-center gap-3 p-3 hover:bg-slate-50">
   248→                            <div className="w-8 h-8 rounded-lg bg-slate-100 flex items-center justify-center text-xs font-bold text-slate-600 shrink-0">
   249→                              {j.urutan}
   250→                            </div>
   251→                            <div className="flex-1 min-w-0">
   252→                              <div className="text-sm font-medium text-slate-800">{j.nama}</div>
   253→                              <div className="text-[10px] text-slate-400">
   254→                                Level: {j.level} • ID: {j.id}
   255→                              </div>
   256→                            </div>
   257→                            <div className="flex items-center gap-2 shrink-0">
   258→                              <span className="inline-flex items-center gap-1 text-[10px] text-slate-500 bg-slate-100 px-2 py-0.5 rounded-full">
   259→                                <Users className="w-3 h-3" />
   260→                                {j._count?.pengurus || 0}
   261→                              </span>
   262→                              <span className={`px-2 py-0.5 rounded-full text-[10px] font-semibold ${
   263→                                j.status === "Aktif" ? "bg-emerald-100 text-emerald-700" : "bg-slate-100 text-slate-500"
   264→                              }`}>
   265→                                {j.status}
   266→                              </span>
   267→                              <button
   268→                                onClick={() => {
   269→                                  setEditingItem(j);
   270→                                  setShowFormDialog(true);
   271→                                }}
   272→                                className="p-1.5 text-slate-400 hover:text-blue-600 hover:bg-blue-50 rounded"
   273→                                title="Edit"
   274→                              >
   275→                                <Edit className="w-4 h-4" />
   276→                              </button>
   277→                              <button
   278→                                onClick={() => handleToggleStatus(j)}
   279→                                className="p-1.5 text-slate-400 hover:text-amber-600 hover:bg-amber-50 rounded"
   280→                                title={j.status === "Aktif" ? "Nonaktifkan" : "Aktifkan"}
   281→                              >
   282→                                <AlertCircle className="w-4 h-4" />
   283→                              </button>
   284→                              <button
   285→                                onClick={() => handleDelete(j.id, j.nama, j.bidang)}
   286→                                className="p-1.5 text-slate-400 hover:text-rose-600 hover:bg-rose-50 rounded"
   287→                                title="Hapus"
   288→                              >
   289→                                <Trash2 className="w-4 h-4" />
   290→                              </button>
   291→                            </div>
   292→                          </div>
   293→                        ))}
   294→                      </div>
   295→                    </motion.div>
   296→                  )}
   297→                </AnimatePresence>
   298→              </div>
   299→            );
   300→          })
   301→        )}
   302→      </div>
   303→
   304→      {/* Form Dialog — Tambah/Edit Jabatan */}
   305→      <JabatanFormDialog
   306→        open={showFormDialog}
   307→        data={editingItem}
   308→        existingBidang={bidangNames}
   309→        onClose={() => {
   310→          setShowFormDialog(false);
   311→          setEditingItem(null);
   312→        }}
   313→        onSaved={() => {
   314→          setShowFormDialog(false);
   315→          setEditingItem(null);
   316→          fetchData();
   317→        }}
   318→      />
   319→
   320→      {/* Add Bidang Dialog */}
   321→      <AddBidangDialog
   322→        open={showAddBidangDialog}
   323→        existingBidang={bidangNames}
   324→        onClose={() => setShowAddBidangDialog(false)}
   325→        onCreated={() => {
   326→          setShowAddBidangDialog(false);
   327→          fetchData();
   328→        }}
   329→      />
   330→    </div>
   331→  );
   332→}
   333→
   334→// ============================================================
   335→// Sub-component: Stat Card
   336→// ============================================================
   337→function StatCard({ label, value, color, bg }: { label: string; value: number; color: string; bg: string }) {
   338→  return (
   339→    <div className={`rounded-xl p-4 ${bg} border border-white`}>
   340→      <div className={`text-2xl font-extrabold ${color}`}>{value}</div>
   341→      <div className="text-[10px] text-slate-600 mt-0.5">{label}</div>
   342→    </div>
   343→  );
   344→}
   345→
   346→// ============================================================
   347→// Sub-component: Jabatan Form Dialog (Tambah/Edit Jabatan)
   348→// ============================================================
   349→function JabatanFormDialog({
   350→  open,
   351→  data,
   352→  existingBidang,
   353→  onClose,
   354→  onSaved,
   355→}: {
   356→  open: boolean;
   357→  data: any;
   358→  existingBidang: string[];
   359→  onClose: () => void;
   360→  onSaved: () => void;
   361→}) {
   362→  const [form, setForm] = useState({
   363→    nama: "",
   364→    bidang: "Pengurus Harian",
   365→    level: "Nasional",
   366→    urutan: 1,
   367→    status: "Aktif",
   368→  });
   369→  const [saving, setSaving] = useState(false);
   370→  const [error, setError] = useState("");
   371→  const [newBidangMode, setNewBidangMode] = useState(false);
   372→
   373→  useEffect(() => {
   374→    if (data) {
   375→      setForm({
   376→        nama: data.nama,
   377→        bidang: data.bidang,
   378→        level: data.level,
   379→        urutan: data.urutan,
   380→        status: data.status,
   381→      });
   382→      setNewBidangMode(false);
   383→    } else {
   384→      setForm({
   385→        nama: "",
   386→        bidang: existingBidang[0] || "Pengurus Harian",
   387→        level: "Nasional",
   388→        urutan: 1,
   389→        status: "Aktif",
   390→      });
   391→      setNewBidangMode(false);
   392→    }
   393→    setError("");
   394→  }, [data, open, existingBidang]);
   395→
   396→  const handleSave = async () => {
   397→    if (!form.nama.trim()) {
   398→      setError("Nama jabatan wajib diisi");
   399→      return;
   400→    }
   401→    if (!form.bidang.trim()) {
   402→      setError("Bidang wajib diisi");
   403→      return;
   404→    }
   405→    setSaving(true);
   406→    setError("");
   407→    try {
   408→      const method = data ? "PATCH" : "POST";
   409→      const url = data ? `/api/jabatan/${data.id}` : "/api/jabatan";
   410→      const res = await fetch(url, {
   411→        method,
   412→        headers: { "Content-Type": "application/json" },
   413→        body: JSON.stringify(form),
   414→      });
   415→      const json = await res.json();
   416→      if (!json.success) {
   417→        throw new Error(json.error);
   418→      }
   419→      toast.success(json.message);
   420→      onSaved();
   421→    } catch (e: any) {
   422→      setError(e.message);
   423→    } finally {
   424→      setSaving(false);
   425→    }
   426→  };
   427→
   428→  return (
   429→    <AnimatePresence>
   430→      {open && (
   431→        <motion.div
   432→          initial={{ opacity: 0 }}
   433→          animate={{ opacity: 1 }}
   434→          exit={{ opacity: 0 }}
   435→          onClick={onClose}
   436→          className="fixed inset-0 z-[300] bg-blue-950/90 backdrop-blur-md flex items-center justify-center p-4"
   437→        >
   438→          <motion.div
   439→            initial={{ scale: 0.95, y: 20 }}
   440→            animate={{ scale: 1, y: 0 }}
   441→            exit={{ scale: 0.95, y: 20 }}
   442→            onClick={(e) => e.stopPropagation()}
   443→            className="bg-white rounded-3xl shadow-2xl max-w-md w-full"
   444→          >
   445→            <div className="relative bg-gradient-to-r from-blue-600 to-sky-500 p-5 text-white">
   446→              <button onClick={onClose} className="absolute top-4 right-4 w-8 h-8 rounded-full bg-white/20 hover:bg-white/30 flex items-center justify-center">
   447→                <X className="w-4 h-4" />
   448→              </button>
   449→              <div className="flex items-center gap-3">
   450→                <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center">
   451→                  <Award className="w-5 h-5" />
   452→                </div>
   453→                <div>
   454→                  <h2 className="text-lg font-bold">{data ? "Edit" : "Tambah"} Jabatan</h2>
   455→                  <p className="text-xs text-blue-100">Isi data jabatan dengan benar</p>
   456→                </div>
   457→              </div>
   458→            </div>
   459→
   460→            <div className="p-6 space-y-4">
   461→              {error && (
   462→                <div className="bg-rose-50 border border-rose-200 text-rose-700 text-xs px-3 py-2 rounded-lg">
   463→                  ⚠️ {error}
   464→                </div>
   465→              )}
   466→
   467→              {/* Bidang */}
   468→              <div>
   469→                <label className="block text-xs font-semibold text-slate-700 mb-1.5">Bidang *</label>
   470→                {!newBidangMode ? (
   471→                  <div className="flex gap-2">
   472→                    <select
   473→                      value={form.bidang}
   474→                      onChange={(e) => setForm({ ...form, bidang: e.target.value })}
   475→                      className="flex-1 px-3 py-2 text-sm border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   476→                    >
   477→                      {existingBidang.length === 0 && <option value="Pengurus Harian">Pengurus Harian</option>}
   478→                      {existingBidang.map((b) => <option key={b} value={b}>{b}</option>)}
   479→                    </select>
   480→                    <button
   481→                      type="button"
   482→                      onClick={() => {
   483→                        setNewBidangMode(true);
   484→                        setForm({ ...form, bidang: "" });
   485→                      }}
   486→                      className="px-3 py-2 text-xs font-semibold text-violet-600 border border-violet-200 rounded-lg hover:bg-violet-50 whitespace-nowrap"
   487→                    >
   488→                      + Bidang Baru
   489→                    </button>
   490→                  </div>
   491→                ) : (
   492→                  <div className="flex gap-2">
   493→                    <input
   494→                      type="text"
   495→                      value={form.bidang}
   496→                      onChange={(e) => setForm({ ...form, bidang: e.target.value })}
   497→                      placeholder="Contoh: Divisi Pemberdayaan Pemuda"
   498→                      className="flex-1 px-3 py-2 text-sm border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   499→                    />
   500→                    <button
   501→                      type="button"
   502→                      onClick={() => {
   503→                        setNewBidangMode(false);
   504→                        setForm({ ...form, bidang: existingBidang[0] || "Pengurus Harian" });
   505→                      }}
   506→                      className="px-3 py-2 text-xs font-semibold text-slate-500 border border-slate-200 rounded-lg hover:bg-slate-50 whitespace-nowrap"
   507→                    >
   508→                      Pilih Existing
   509→                    </button>
   510→                  </div>
   511→                )}
   512→              </div>
   513→
   514→              {/* Nama Jabatan */}
   515→              <div>
   516→                <label className="block text-xs font-semibold text-slate-700 mb-1.5">Nama Jabatan *</label>
   517→                <input
   518→                  type="text"
   519→                  value={form.nama}
   520→                  onChange={(e) => setForm({ ...form, nama: e.target.value })}
   521→                  placeholder="Contoh: Ketua Divisi, Sekretaris, Anggota"
   522→                  className="w-full px-3 py-2 text-sm border border-slate-200 rounded-lg focus:border-blue-500 outline-none"

   585→// ============================================================
   586→// Sub-component: Add Bidang Dialog — create new bidang with 3 default jabatan
   587→// ============================================================
   588→function AddBidangDialog({
   589→  open,
   590→  existingBidang,
   591→  onClose,
   592→  onCreated,
   593→}: {
   594→  open: boolean;
   595→  existingBidang: string[];
   596→  onClose: () => void;
   597→  onCreated: () => void;
   598→}) {
   599→  const [bidangName, setBidangName] = useState("");
   600→  const [levels, setLevels] = useState({ Nasional: true, Provinsi: true, Kabupaten: true });
   601→  const [saving, setSaving] = useState(false);
   602→  const [error, setError] = useState("");
   603→
   604→  useEffect(() => {
   605→    if (open) {
   606→      setBidangName("");
   607→      setLevels({ Nasional: true, Provinsi: true, Kabupaten: true });
   608→      setError("");
   609→    }
   610→  }, [open]);
   611→
   612→  const handleCreate = async () => {
   613→    if (!bidangName.trim()) {
   614→      setError("Nama bidang wajib diisi");
   615→      return;
   616→    }
   617→    if (existingBidang.includes(bidangName.trim())) {
   618→      setError("Bidang dengan nama ini sudah ada");
   619→      return;
   620→    }
   621→    const selectedLevels = Object.keys(levels).filter((l) => (levels as any)[l]);
   622→    if (selectedLevels.length === 0) {
   623→      setError("Pilih minimal 1 level");
   624→      return;
   625→    }
   626→    setSaving(true);
   627→    setError("");
   628→    try {
   629→      // Create 3 default jabatan (Ketua Divisi, Sekretaris Divisi, Anggota) for each selected level
   630→      const defaultJabatan = [
   631→        { nama: "Ketua Divisi", urutan: 1 },
   632→        { nama: "Sekretaris Divisi", urutan: 2 },
   633→        { nama: "Anggota", urutan: 3 },
   634→      ];
   635→      let created = 0;
   636→      for (const level of selectedLevels) {
   637→        for (const dj of defaultJabatan) {
   638→          const res = await fetch("/api/jabatan", {
   639→            method: "POST",
   640→            headers: { "Content-Type": "application/json" },
   641→            body: JSON.stringify({
   642→              nama: dj.nama,
   643→              bidang: bidangName.trim(),
   644→              level,
   645→              urutan: dj.urutan,
   646→            }),
   647→          });
   648→          const json = await res.json();
   649→          if (json.success) created++;
   650→        }
   651→      }
   652→      toast.success(`Bidang "${bidangName.trim()}" dibuat dengan ${created} jabatan di ${selectedLevels.length} level`);
   653→      onCreated();
   654→    } catch (e: any) {
   655→      setError(e.message);
   656→    } finally {
   657→      setSaving(false);
   658→    }
   659→  };
   660→
   661→  return (
   662→    <AnimatePresence>
   663→      {open && (
   664→        <motion.div
   665→          initial={{ opacity: 0 }}
   666→          animate={{ opacity: 1 }}
   667→          exit={{ opacity: 0 }}
   668→          onClick={onClose}
   669→          className="fixed inset-0 z-[300] bg-blue-950/90 backdrop-blur-md flex items-center justify-center p-4"
   670→        >
   671→          <motion.div
   672→            initial={{ scale: 0.95, y: 20 }}
   673→            animate={{ scale: 1, y: 0 }}
   674→            exit={{ scale: 0.95, y: 20 }}
   675→            onClick={(e) => e.stopPropagation()}
   676→            className="bg-white rounded-3xl shadow-2xl max-w-md w-full"
   677→          >
   678→            <div className="relative bg-gradient-to-r from-violet-600 to-purple-500 p-5 text-white">
   679→              <button onClick={onClose} className="absolute top-4 right-4 w-8 h-8 rounded-full bg-white/20 hover:bg-white/30 flex items-center justify-center">
   680→                <X className="w-4 h-4" />
   681→              </button>
   682→              <div className="flex items-center gap-3">
   683→                <div className="w-10 h-10 rounded-xl bg-white/20 flex items-center justify-center">
   684→                  <Plus className="w-5 h-5" />
   685→                </div>
   686→                <div>
   687→                  <h2 className="text-lg font-bold">Tambah Bidang Baru</h2>
   688→                  <p className="text-xs text-violet-100">Otomatis dibuat 3 jabatan: Ketua Divisi, Sekretaris Divisi, Anggota</p>
   689→                </div>
   690→              </div>
   691→            </div>
   692→
   693→            <div className="p-6 space-y-4">
   694→              {error && (
   695→                <div className="bg-rose-50 border border-rose-200 text-rose-700 text-xs px-3 py-2 rounded-lg">
   696→                  ⚠️ {error}
   697→                </div>
   698→              )}
   699→
   700→              <div>
   701→                <label className="block text-xs font-semibold text-slate-700 mb-1.5">Nama Bidang *</label>
   702→                <input
   703→                  type="text"
   704→                  value={bidangName}
   705→                  onChange={(e) => setBidangName(e.target.value)}
   706→                  placeholder="Contoh: Divisi Pemberdayaan Pemuda"
   707→                  className="w-full px-3 py-2 text-sm border border-slate-200 rounded-lg focus:border-violet-500 outline-none"
   708→                />
   709→                <p className="text-[10px] text-slate-400 mt-1">
   710→                  Contoh: "Divisi Pemberdayaan Pemuda", "Divisi Hukum dan Advokasi"
   711→                </p>
   712→              </div>
   713→
   714→              <div>
   715→                <label className="block text-xs font-semibold text-slate-700 mb-1.5">Buat di Level *</label>
   716→                <div className="space-y-2">
   717→                  {LEVELS.map((l) => (
   718→                    <label key={l} className="flex items-center gap-2 p-2 rounded-lg border border-slate-200 hover:bg-slate-50 cursor-pointer">
   719→                      <input
   720→                        type="checkbox"
   721→                        checked={(levels as any)[l]}
   722→                        onChange={(e) => setLevels({ ...levels, [l]: e.target.checked })}
   723→                        className="w-4 h-4 rounded text-violet-600 focus:ring-violet-500"
   724→                      />
   725→                      <span className="text-sm text-slate-700">{l}</span>
   726→                    </label>
   727→                  ))}
   728→                </div>
   729→              </div>
   730→
   731→              <div className="bg-violet-50 border border-violet-100 rounded-lg p-3">
   732→                <p className="text-xs text-violet-700">
   733→                  ℹ️ Akan dibuat <strong>3 jabatan default</strong> (Ketua Divisi, Sekretaris Divisi, Anggota) di setiap level yang dipilih. Anda bisa menambah jabatan lain setelah bidang dibuat.
   734→                </p>
   735→              </div>
   736→            </div>
   737→
   738→            <div className="p-5 border-t border-slate-100 flex justify-end gap-2">
   739→              <button onClick={onClose} className="px-4 py-2 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-lg">
   740→                Batal
   741→              </button>
   742→              <button
   743→                onClick={handleCreate}
   744→                disabled={saving}
   745→                className="inline-flex items-center gap-1.5 px-4 py-2 bg-violet-600 text-white text-sm font-semibold rounded-lg hover:bg-violet-700 disabled:opacity-50"
   746→              >
   747→                {saving ? (
   748→                  <><div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" /> Membuat...</>
   749→                ) : (
   750→                  <><Plus className="w-4 h-4" /> Buat Bidang</>
   751→                )}
   752→              </button>
   753→            </div>
   754→          </motion.div>
   755→        </motion.div>
   756→      )}
   757→    </AnimatePresence>
   758→  );
   759→}
