   558→              {step === 2 && (

   560→                  key="step2"
   561→                  initial={{ opacity: 0, x: 20 }}
   562→                  animate={{ opacity: 1, x: 0 }}
   563→                  exit={{ opacity: 0, x: -20 }}
   564→                  className="space-y-4"
   565→                >
   566→                  <h3 className="text-xl font-bold text-blue-950 mb-4">

   569→
   570→                  <Field label="Email" required>

   573→                      value={form.email}
   574→                      onChange={(e) => update("email", e.target.value)}

   576→                      placeholder="email@contoh.com"
   577→                    />
   578→                  </Field>
   579→
   580→                  <div className="grid sm:grid-cols-2 gap-4">
   581→                    <Field label="Nomor HP" required>
   582→                      <input
   583→                        type="tel"
   584→                        value={form.nomorHP}
   585→                        onChange={(e) => update("nomorHP", e.target.value)}
   586→                        className="form-input"
   587→                        placeholder="08xxxxxxxxxx"
   588→                      />
   589→                    </Field>
   590→                    <Field label="Nomor WhatsApp" required>
   591→                      <input
   592→                        type="tel"
   593→                        value={form.whatsapp}
   594→                        onChange={(e) => update("whatsapp", e.target.value)}

   596→                        placeholder="08xxxxxxxxxx"
   597→                      />
   598→                    </Field>
   599→                  </div>
   600→
   601→                  <Field label="Motivasi Bergabung dengan KIPAN">

   618→                    </p>
   619→                    <div className="grid sm:grid-cols-2 gap-3">

   621→                        { key: "ktp", label: "KTP", icon: IdCard },
   622→                        { key: "foto", label: "Pas Foto", icon: User },
   623→                        { key: "cv", label: "CV/Resume", icon: FileText },

   628→                        const value = (form as any)[doc.key] as string | undefined;
   629→                        return (
   630→                          <div key={doc.key} className={`p-3 rounded-xl border-2 transition-all ${value ? "border-emerald-300 bg-emerald-50/50" : "border-slate-200 bg-white"}`}>
   631→                            <div className="flex items-center gap-2 mb-2">

   633→                                <Icon className={`w-4 h-4 ${value ? "text-emerald-600" : "text-sky-600"}`} />
   634→                              </div>
   635→                              <span className="text-xs font-semibold text-slate-700">{doc.label}</span>

   637→                            </div>
   638→                            <input
   639→                              type="file"
   640→                              accept="image/*,.pdf"
   641→                              onChange={async (e) => {
   642→                                const file = e.target.files?.[0];
   643→                                if (!file) return;
   644→                                if (file.size > 1024 * 1024 * 2) {
   645→                                  alert("File maksimal 2MB");
   646→                                  return;
   647→                                }
   648→                                const reader = new FileReader();
   649→                                reader.onload = () => {
   650→                                  setForm((prev) => ({ ...prev, [doc.key]: reader.result as string }));
   651→                                };
   652→                                reader.readAsDataURL(file);
   653→                              }}
   654→                              className="w-full text-xs border border-slate-200 rounded-lg px-2 py-1.5 file:mr-2 file:py-0.5 file:px-2 file:rounded file:border-0 file:text-xs file:font-medium file:bg-sky-50 file:text-sky-700 hover:file:bg-sky-100"
   655→                            />
   656→                            {value && (
   657→                              <div className="flex items-center gap-2 mt-1.5">
   658→                                <button
   659→                                  type="button"
   660→                                  onClick={() => {
   661→                                    const w = window.open();
   662→                                    if (w) {
   663→                                      if (value.startsWith("data:image/")) {
   664→                                        w.document.write(`<html><head><title>${doc.label}</title></head><body style="margin:0;display:flex;justify-content:center;align-items:center;min-height:100vh;background:#1e293b"><img src="${value}" style="max-width:100%;max-height:100vh;object-fit:contain" /></body></html>`);
   665→                                      } else if (value.startsWith("data:application/pdf")) {
   666→                                        w.document.write(`<html><head><title>${doc.label}</title></head><body style="margin:0"><iframe src="${value}" style="width:100vw;height:100vh;border:0"></iframe></body></html>`);
   667→                                      } else {
   668→                                        w.document.write(`<html><head><title>${doc.label}</title></head><body style="margin:0;display:flex;justify-content:center;align-items:center;min-height:100vh"><a href="${value}" download="${doc.label}" style="padding:12px 24px;background:#0ea5e9;color:white;text-decoration:none;border-radius:8px">Download ${doc.label}</a></body></html>`);
   669→                                      }
   670→                                      w.document.close();
   671→                                    }
   672→                                  }}

   680→                                  className="text-[10px] text-rose-500 hover:text-rose-700 underline"

   682→                                  Hapus
   683→                                </button>
   684→                              </div>
   685→                            )}
   686→                          </div>
   687→                        );
   688→                      })}
   689→                    </div>
   690→                  </div>
   691→                </motion.div>
   692→              )}
   693→
   694→              {/* STEP 3: Persyaratan */}
   695→              {step === 3 && (
   696→                <motion.div
   697→                  key="step3"
   698→                  initial={{ opacity: 0, x: 20 }}
   699→                  animate={{ opacity: 1, x: 0 }}
   700→                  exit={{ opacity: 0, x: -20 }}
   701→                  className="space-y-4"
   702→                >
   703→                  <h3 className="text-xl font-bold text-blue-950 mb-2">
   704→                    Persyaratan Kepengurusan
   705→                  </h3>
   706→                  <p className="text-sm text-slate-600 mb-4">
   707→                    Centang seluruh persyaratan di bawah ini untuk melanjutkan.
   708→                  </p>
   709→
   710→                  <div className="space-y-3">
   711→                    {persyaratan.map((p, idx) => {
   712→                      const Icon = PERSYARATAN_ICONS[p.icon] || CheckCircle2;
   713→                      const checked = form.persyaratan[idx];
   714→                      return (
   715→                        <motion.label
   716→                          key={idx}
   717→                          initial={{ opacity: 0, y: 10 }}
   718→                          animate={{ opacity: 1, y: 0 }}
   719→                          transition={{ delay: idx * 0.05 }}
   720→                          className={`flex items-start gap-3 p-4 rounded-2xl border-2 cursor-pointer transition-all ${
   721→                            checked
   722→                              ? "border-sky-400 bg-sky-50"
   723→                              : "border-slate-200 hover:border-sky-300"
   724→                          }`}
   725→                        >
   726→                          <button
   727→                            type="button"
   728→                            onClick={() => {
   729→                              const next = [...form.persyaratan];
   730→                              next[idx] = !next[idx];
   731→                              update("persyaratan", next);
   732→                            }}
   733→                            className={`shrink-0 w-6 h-6 rounded-md border-2 flex items-center justify-center transition-all ${
   734→                              checked
   735→                                ? "bg-sky-500 border-sky-500"
   736→                                : "border-slate-300"
   737→                            }`}
   738→                          >
   739→                            {checked && <CheckCircle2 className="w-4 h-4 text-white" />}
   740→                          </button>
   741→                          <div className="w-10 h-10 rounded-xl bg-sky-100 flex items-center justify-center shrink-0">
   742→                            <Icon className="w-5 h-5 text-blue-600" />
   743→                          </div>
   744→                          <div className="flex-1">
   745→                            <div className="font-semibold text-slate-800 text-sm">
   746→                              {p.title}
   747→                            </div>
   748→                            <div className="text-xs text-slate-500 mt-0.5">
   749→                              {p.desc}
   750→                            </div>
   751→                          </div>
   752→                        </motion.label>
   753→                      );
   754→                    })}
   755→                  </div>
   756→
   757→                  {!allPersyaratanChecked && (
   758→                    <p className="text-xs text-amber-600 bg-amber-50 border border-amber-200 rounded-lg p-3">
   759→                      ⚠️ Semua persyaratan harus dicentang untuk melanjutkan ke tahap konfirmasi.
   760→                    </p>
   761→                  )}
   762→                </motion.div>
   763→              )}
   764→
   765→              {/* STEP 4: Konfirmasi */}
   766→              {step === 4 && (
   767→                <motion.div
   768→                  key="step4"
   769→                  initial={{ opacity: 0, x: 20 }}
   770→                  animate={{ opacity: 1, x: 0 }}
   771→                  exit={{ opacity: 0, x: -20 }}
   772→                  className="space-y-4"
   773→                >
   774→                  <h3 className="text-xl font-bold text-blue-950 mb-4">
   775→                    Konfirmasi Data
   776→                  </h3>
   777→
   778→                  <div className="bg-sky-50 border border-sky-200 rounded-2xl p-5 space-y-3">
   779→                    <div className="text-xs font-bold text-blue-700 uppercase tracking-wider">
   780→                      Data Diri
   781→                    </div>
   782→                    <Row label="Nama" value={form.namaLengkap} />
   783→                    <Row label="NIK" value={form.nik} />
   784→                    <Row label="TTL" value={`${form.tempatLahir}, ${form.tanggalLahir}`} />
   785→                    <Row label="Jenis Kelamin" value={form.jenisKelamin === "L" ? "Laki-laki" : form.jenisKelamin === "P" ? "Perempuan" : "-"} />
   786→                    <Row label="Alamat" value={`${form.alamat}, ${form.kecamatan}, ${form.kabupaten}, ${form.provinsi} ${form.kodePos}`} />
   787→                  </div>
   788→
   789→                  <div className="bg-sky-50 border border-sky-200 rounded-2xl p-5 space-y-3">
   790→                    <div className="text-xs font-bold text-blue-700 uppercase tracking-wider">
   791→                      Kontak
   792→                    </div>
   793→                    <Row label="Email" value={form.email} />
   794→                    <Row label="HP" value={form.nomorHP} />
   795→                    <Row label="WhatsApp" value={form.whatsapp} />
   796→                  </div>
   797→
   798→                  <div className="bg-amber-50 border border-amber-200 rounded-2xl p-4">
   799→                    <p className="text-xs text-amber-800 leading-relaxed">
   800→                      <strong>Pernyataan:</strong> Saya menyatakan bahwa data yang
   801→                      saya isi adalah benar dan dapat dipertanggungjawabkan. Saya
   802→                      bersedia mengikuti seluruh rangkaian verifikasi dan pelatihan
   803→                      yang diselenggarakan KIPAN.
   804→                    </p>
   805→                  </div>
   806→                </motion.div>
   807→              )}
   808→            </AnimatePresence>
   809→
   810→            {/* Navigation buttons */}
   811→            <div className="flex items-center justify-between mt-8 pt-6 border-t border-slate-100">
   812→              <button
   813→                onClick={prevStep}
   814→                disabled={step === 1}
   815→                className="inline-flex items-center gap-2 px-5 py-2.5 text-sm font-medium text-slate-600 hover:text-blue-700 disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
   816→              >
   817→                <ChevronLeft className="w-4 h-4" />
   818→                Kembali
   819→              </button>
   820→
   821→              {step < STEPS.length ? (
   822→                <button
   823→                  onClick={nextStep}
   824→                  disabled={(step === 3 && !allPersyaratanChecked)}
   825→                  className="inline-flex items-center gap-2 px-6 py-3 bg-gradient-to-r from-sky-500 to-blue-600 text-white font-semibold rounded-full shadow-lg shadow-sky-500/30 hover:shadow-xl hover:-translate-y-0.5 transition-all disabled:opacity-50 disabled:cursor-not-allowed disabled:hover:translate-y-0"
   826→                >

   828→                  <ChevronRight className="w-4 h-4" />

   831→                <div className="flex flex-col items-end gap-2">
   832→                  {submitError && (
   833→                    <div className="text-xs text-rose-600 bg-rose-50 border border-rose-200 rounded-lg px-3 py-2 max-w-xs">
   834→                      ⚠️ {submitError}
   835→                    </div>

   837→                  <button
   838→                    onClick={handleSubmit}
   839→                    disabled={submitting}

   841→                  >
   842→                    {submitting ? (
   843→                      <>
   844→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   845→                        Mengirim...
   846→                      </>
   847→                    ) : (
   848→                      <>
   849→                        <Send className="w-4 h-4" />
   850→                        Kirim Pendaftaran
   851→                      </>
   852→                    )}
   853→                  </button>
   854→                </div>
   855→              )}
   856→            </div>
   857→          </motion.div>
   858→        </div>
   859→      </div>
   860→
   861→      {/* Inline style for form inputs */}
   862→      <style jsx global>{`
   863→        .form-input {
   864→          width: 100%;
   865→          padding: 0.625rem 0.875rem;
   866→          font-size: 0.875rem;
   867→          border-radius: 0.5rem;
   868→          border: 1px solid #e2e8f0;
   869→          background: white;
   870→          outline: none;
   871→          transition: all 0.2s;
   872→        }
   873→        .form-input:focus {
   874→          border-color: #10b981;
   875→          box-shadow: 0 0 0 3px rgba(16, 185, 129, 0.15);
   876→        }
   877→      `}</style>
   878→    </section>
   879→  );
   880→}
   881→
   882→function Field({
   883→  label,
   884→  required,
   885→  children,
   886→}: {
   887→  label: string;
   888→  required?: boolean;
   889→  children: React.ReactNode;
   890→}) {
   891→  return (
   892→    <label className="block">
   893→      <span className="block text-xs font-semibold text-slate-700 mb-1.5">
   894→        {label} {required && <span className="text-rose-500">*</span>}
   895→      </span>
   896→      {children}
   897→    </label>
   898→  );
   899→}
   900→
   901→function Row({ label, value }: { label: string; value: string }) {
   902→  return (
   903→    <div className="flex items-start gap-3 text-sm">
   904→      <span className="text-slate-500 min-w-[100px]">{label}</span>
   905→      <span className="text-slate-800 font-medium flex-1">{value || "-"}</span>
   906→    </div>
   907→  );
   908→}
