     1→"use client";
     2→
     3→import { useState, useEffect, useMemo } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  MapPin,
     7→  Building2,
     8→  UserCog,
     9→  Plus,
    10→  Search,
    11→  RefreshCw,
    12→  Download,
    13→  FileText,
    14→  Eye,
    15→  Edit,
    16→  MoreVertical,
    17→  ChevronLeft,
    18→  ChevronRight,
    19→  ChevronsUpDown,
    20→  ChevronUp,
    21→  ChevronDown,
    22→  MapPinned,
    23→  Ban,
    24→  Trash2,
    25→  ArrowUpDown,
    26→  Inbox,
    27→  CheckCircle2,
    28→} from "lucide-react";
    29→import { PROVINSI_LIST, KABUPATEN_LIST } from "@/lib/admin-data";
    30→import WilayahDetailDialog from "./WilayahDetailDialog";
    31→import WilayahFormDialog, { type WilayahFormData } from "./WilayahFormDialog";
    32→import { exportWilayahPdf } from "@/lib/pdf-export";
    33→import { toast } from "sonner";
    34→
    35→type SortDir = "asc" | "desc" | null;
    36→
    37→export default function WilayahPage({
    38→  onNavigate,
    39→  userRole = "SUPER_ADMIN",
    40→}: {
    41→  onNavigate?: (page: string, filter?: Record<string, string>) => void;
    42→  userRole?: string;
    43→}) {
    44→  const [tab, setTab] = useState<"provinsi" | "kabupaten">("provinsi");
    45→  const [search, setSearch] = useState("");
    46→  const [statusFilter, setStatusFilter] = useState("Semua");
    47→  const [provinsiFilter, setProvinsiFilter] = useState("Semua");
    48→  const [loading, setLoading] = useState(true);
    49→  const [sortBy, setSortBy] = useState<string | null>(null);
    50→  const [sortDir, setSortDir] = useState<SortDir>(null);
    51→  const [page, setPage] = useState(1);
    52→  const [rowsPerPage, setRowsPerPage] = useState(10);
    53→  const [detailId, setDetailId] = useState<number | null>(null);
    54→  const [detailType, setDetailType] = useState<"provinsi" | "kabupaten">("provinsi");
    55→  const [actionMenuId, setActionMenuId] = useState<number | null>(null);
    56→  const [showFormDialog, setShowFormDialog] = useState(false);
    57→  const [formData, setFormData] = useState<WilayahFormData | null>(null);
    58→  const [apiProvinsi, setApiProvinsi] = useState<any[]>([]);
    59→  const [apiKabupaten, setApiKabupaten] = useState<any[]>([]);
    60→  const [useApiData, setUseApiData] = useState(false);
    61→  const [totalAllPengurus, setTotalAllPengurus] = useState(0);
    62→
    63→  // Fetch from API
    64→  const fetchData = async () => {
    65→    setLoading(true);
    66→    try {
    67→      const [wilRes, pengRes] = await Promise.all([
    68→        fetch("/api/wilayah", { cache: "no-store" }),
    69→        fetch("/api/pengurus", { cache: "no-store" }),
    70→      ]);
    71→      const wilJson = await wilRes.json();
    72→      const pengJson = await pengRes.json();
    73→      if (wilJson.success) {
    74→        setApiProvinsi(wilJson.data.provinsi || []);
    75→        setApiKabupaten(wilJson.data.kabupaten || []);
    76→        setUseApiData(true);
    77→      }
    78→      if (pengJson.success) {
    79→        setTotalAllPengurus(pengJson.total || 0);
    80→      }
    81→    } catch (e) {
    82→      console.error("Failed to fetch wilayah:", e);
    83→    } finally {
    84→      setLoading(false);
    85→    }
    86→  };
    87→
    88→  // Auto-fill ketua wilayah dari pengurus dengan jabatan Ketua Umum
    89→  const autoFillKetua = async () => {
    90→    try {
    91→      const res = await fetch("/api/wilayah/auto-fill-ketua", { method: "POST" });
    92→      const json = await res.json();
    93→      if (json.success) {
    94→        toast.success(json.message);
    95→        fetchData();
    96→      } else {
    97→        toast.error(json.error || "Gagal auto-fill ketua");
    98→      }
    99→    } catch (e) {
   100→      toast.error("Gagal auto-fill ketua");
   101→    }
   102→  };
   103→
   104→  // Initial load — auto-fill ketua dulu, lalu fetch data
   105→  useEffect(() => {
   106→    fetch("/api/wilayah/auto-fill-ketua", { method: "POST" })
   107→      .then(() => fetchData())
   108→      .catch(() => fetchData());
   109→  }, []);
   110→
   111→  // Permission check
   112→  const canCreateProvinsi = userRole === "SUPER_ADMIN";
   113→  const canCreateKabupaten = ["SUPER_ADMIN", "ADMIN_NASIONAL", "ADMIN_PROVINSI"].includes(userRole);
   114→  const canDelete = userRole === "SUPER_ADMIN";
   115→  const canEdit = ["SUPER_ADMIN", "ADMIN_NASIONAL", "ADMIN_PROVINSI"].includes(userRole);
   116→
   117→  // Use API data if available, fallback to mock data
   118→  const provData = useApiData ? apiProvinsi.map((p: any) => ({
   119→    id: p.id,
   120→    kode: p.kode,
   121→    nama: p.nama,
   122→    status: p.status,
   123→    ketua: p.ketua || "",
   124→    jumlahKabupaten: p._count?.kabupaten || 0,
   125→    jumlahPengurus: p._count?.pengurus || 0,
   126→    createdAt: p.createdAt,
   127→    updatedAt: p.updatedAt,
   128→  })) : PROVINSI_LIST;
   129→
   130→  const kabData = useApiData ? apiKabupaten.map((k: any) => ({
   131→    id: k.id,
   132→    kode: k.kode,
   133→    nama: k.nama,
   134→    provinsiNama: k.provinsi?.nama || "",
   135→    status: k.status,
   136→    ketua: k.ketua || "",
   137→    jumlahPengurus: k._count?.pengurus || 0,
   138→    createdAt: k.createdAt,
   139→    updatedAt: k.updatedAt,
   140→  })) : KABUPATEN_LIST;
   141→
   142→  // Stat cards — use API data for consistency with dashboard
   143→  const statCards = [
   144→    { label: "Total Provinsi", value: provData.length, total: 38, icon: MapPin, color: "from-blue-500 to-sky-500", targetPage: null },
   145→    { label: "Total Kabupaten/Kota", value: kabData.length, total: 514, icon: Building2, color: "from-sky-500 to-cyan-500", targetPage: null },
   146→    { label: "Total Pengurus", value: totalAllPengurus, total: null, icon: UserCog, color: "from-violet-500 to-purple-500", targetPage: "pengurus" },
   147→  ];
   148→
   149→  // Filter data
   150→  const filteredProv = useMemo(() => {
   151→    let result = provData.filter((p: any) => {
   152→      const matchSearch = p.nama.toLowerCase().includes(search.toLowerCase()) ||
   153→        p.kode.toLowerCase().includes(search.toLowerCase()) ||
   154→        (p.ketua || "").toLowerCase().includes(search.toLowerCase());
   155→      const matchStatus = statusFilter === "Semua" || p.status === statusFilter;
   156→      return matchSearch && matchStatus;
   157→    });
   158→    if (sortBy && sortDir) {
   159→      result = [...result].sort((a: any, b: any) => {
   160→        const aVal = a[sortBy];
   161→        const bVal = b[sortBy];
   162→        if (typeof aVal === "string") {
   163→          return sortDir === "asc" ? aVal.localeCompare(bVal) : bVal.localeCompare(aVal);
   164→        }
   165→        return sortDir === "asc" ? aVal - bVal : bVal - aVal;
   166→      });
   167→    }
   168→    return result;
   169→  }, [provData, search, statusFilter, sortBy, sortDir]);
   170→
   171→  const filteredKab = useMemo(() => {
   172→    let result = kabData.filter((k: any) => {
   173→      const matchSearch = k.nama.toLowerCase().includes(search.toLowerCase()) ||
   174→        k.kode.toLowerCase().includes(search.toLowerCase());
   175→      const matchStatus = statusFilter === "Semua" || k.status === statusFilter;
   176→      const matchProv = provinsiFilter === "Semua" || k.provinsiNama === provinsiFilter;
   177→      return matchSearch && matchStatus && matchProv;
   178→    });
   179→    if (sortBy && sortDir) {
   180→      result = [...result].sort((a: any, b: any) => {
   181→        const aVal = a[sortBy];
   182→        const bVal = b[sortBy];
   183→        if (typeof aVal === "string") {
   184→          return sortDir === "asc" ? aVal.localeCompare(bVal) : bVal.localeCompare(aVal);
   185→        }
   186→        return sortDir === "asc" ? aVal - bVal : bVal - aVal;
   187→      });
   188→    }
   189→    return result;
   190→  }, [kabData, search, statusFilter, provinsiFilter, sortBy, sortDir]);
   191→
   192→  const currentData = tab === "provinsi" ? filteredProv : filteredKab;
   193→  const totalData = currentData.length;
   194→  const totalPages = Math.ceil(totalData / rowsPerPage) || 1;
   195→  const currentPage = Math.min(page, totalPages);
   196→  const startIdx = (currentPage - 1) * rowsPerPage;
   197→  const endIdx = Math.min(startIdx + rowsPerPage, totalData);
   198→  const pageData = currentData.slice(startIdx, endIdx);
   199→
   200→  // Summary stats from filtered data
   201→  const filterSummary = tab === "provinsi" ? {
   202→    total: filteredProv.length,
   203→    aktif: filteredProv.filter((p: any) => p.status === "Aktif").length,
   204→    pembentukan: filteredProv.filter((p: any) => p.status === "Pembentukan").length,
   205→    nonaktif: filteredProv.filter((p: any) => p.status === "Nonaktif").length,
   206→    totalPengurus: filteredProv.reduce((a: number, b: any) => a + (b.jumlahPengurus || 0), 0),
   207→    totalKabupaten: filteredProv.reduce((a: number, b: any) => a + (b.jumlahKabupaten || 0), 0),
   208→  } : {
   209→    total: filteredKab.length,
   210→    aktif: filteredKab.filter((k: any) => k.status === "Aktif").length,
   211→    pembentukan: filteredKab.filter((k: any) => k.status === "Pembentukan").length,
   212→    nonaktif: filteredKab.filter((k: any) => k.status === "Nonaktif").length,
   213→    totalPengurus: filteredKab.reduce((a: number, b: any) => a + (b.jumlahPengurus || 0), 0),
   214→    totalKabupaten: 0,
   215→  };
   216→  const hasActiveFilters = search !== "" || statusFilter !== "Semua" || provinsiFilter !== "Semua";
   217→
   218→  const handleSort = (col: string) => {
   219→    if (sortBy === col) {
   220→      setSortDir(sortDir === "asc" ? "desc" : sortDir === "desc" ? null : "asc");
   221→      if (sortDir === "desc") {
   222→        setSortBy(null);
   223→        setSortDir(null);
   224→      }
   225→    } else {
   226→      setSortBy(col);
   227→      setSortDir("asc");
   228→    }
   229→  };
   230→
   231→  const getSortIcon = (col: string) => {
   232→    if (sortBy !== col) return <ArrowUpDown className="w-3 h-3 text-slate-300" />;
   233→    if (sortDir === "asc") return <ChevronUp className="w-3 h-3 text-blue-600" />;
   234→    if (sortDir === "desc") return <ChevronDown className="w-3 h-3 text-blue-600" />;
   235→    return <ArrowUpDown className="w-3 h-3 text-slate-300" />;
   236→  };
   237→
   238→  const switchTab = (newTab: "provinsi" | "kabupaten") => {
   239→    if (newTab === tab) return;
   240→    setLoading(true);
   241→    setTab(newTab);
   242→    setPage(1);
   243→    setTimeout(() => setLoading(false), 600);
   244→  };
   245→
   246→  const handleRefresh = () => {
   247→    setLoading(true);
   248→    setSearch("");
   249→    setStatusFilter("Semua");
   250→    setProvinsiFilter("Semua");
   251→    setSortBy(null);
   252→    setSortDir(null);
   253→    setPage(1);
   254→    setTimeout(() => setLoading(false), 600);
   255→  };
   256→
   257→  const statusBadge = (status: string) => {
   258→    const styles: Record<string, string> = {
   259→      Aktif: "bg-emerald-100 text-emerald-700 border-emerald-200",
   260→      Pembentukan: "bg-amber-100 text-amber-700 border-amber-200",
   261→      Nonaktif: "bg-slate-100 text-slate-600 border-slate-200",
   262→      Dibekukan: "bg-rose-100 text-rose-700 border-rose-200",
   263→    };
   264→    return styles[status] || "bg-slate-100 text-slate-600 border-slate-200";
   265→  };
   266→
   267→  const showDetail = (item: any, type: "provinsi" | "kabupaten") => {
   268→    setDetailType(type);
   269→    setDetailId(item.id);
   270→    setActionMenuId(null);
   271→  };
   272→
   273→  const handleSave = async (data: WilayahFormData) => {
   274→    const method = data.id ? "PUT" : "POST";
   275→    const res = await fetch("/api/wilayah", {
   276→      method,
   277→      headers: { "Content-Type": "application/json" },
   278→      body: JSON.stringify(data),
   279→    });
   280→    const json = await res.json();
   281→    if (!json.success) throw new Error(json.error);
   282→    toast.success(json.message);
   283→    fetchData();
   284→  };
   285→
   286→  const handleDelete = async (item: any) => {
   287→    // Soft delete: ubah status menjadi Nonaktif (data tidak hilang)
   288→    const res = await fetch("/api/wilayah", {
   289→      method: "PUT",
   290→      headers: { "Content-Type": "application/json" },
   291→      body: JSON.stringify({ id: item.id, type: tab, nama: item.nama, status: "Nonaktif", ketua: item.ketua }),
   292→    });
   293→    const json = await res.json();
   294→    if (json.success) {
   295→      toast.success(`${item.nama} dinonaktifkan (data tidak dihapus)`);
   296→      fetchData();
   297→    } else {
   298→      toast.error(json.error || "Gagal menonaktifkan");
   299→    }
   300→  };
   301→
   302→  const handleToggleStatus = async (item: any) => {
   303→    const newStatus = item.status === "Aktif" ? "Nonaktif" : "Aktif";
   304→    const res = await fetch("/api/wilayah", {
   305→      method: "PUT",
   306→      headers: { "Content-Type": "application/json" },
   307→      body: JSON.stringify({ id: item.id, type: tab, nama: item.nama, status: newStatus, ketua: item.ketua }),
   308→    });
   309→    const json = await res.json();
   310→    if (json.success) {
   311→      toast.success(`Status diubah menjadi ${newStatus}`);
   312→      fetchData();
   313→    } else {
   314→      toast.error(json.error || "Gagal mengubah status");
   315→    }
   316→  };
   317→
   318→  const openEditForm = (item: any) => {
   319→    setFormData({
   320→      id: item.id,
   321→      type: tab,
   322→      kode: item.kode,
   323→      nama: item.nama,
   324→      status: item.status,
   325→      ketua: item.ketua || "",
   326→      provinsiId: tab === "kabupaten" ? String(apiProvinsi.find((p: any) => p.nama === item.provinsiNama)?.id || "") : undefined,
   327→      masterProvinsiKode: tab === "kabupaten" ? apiProvinsi.find((p: any) => p.nama === item.provinsiNama)?.kode : undefined,
   328→      masterKabupatenKode: item.kode,
   329→    });
   330→    setShowFormDialog(true);
   331→  };
   332→
   333→  const actions = [
   334→    { label: "Detail", icon: Eye, action: (item: any) => showDetail(item, tab), show: true },
   335→    { label: "Edit", icon: Edit, action: (item: any) => {
   336→      setActionMenuId(null);
   337→      openEditForm(item);
   338→    }, show: canEdit },
   339→    { label: "Kelola Pengurus", icon: UserCog, action: (item: any) => {
   340→      setActionMenuId(null);
   341→      const filter: Record<string, string> = {};
   342→      if (tab === "provinsi") filter.provinsiNama = item.nama;
   343→      else filter.kabupatenNama = item.nama;
   344→      onNavigate?.("pengurus", filter);
   345→    }, show: true },
   346→    { label: "Ubah Status", icon: Ban, action: (item: any) => {
   347→      setActionMenuId(null);
   348→      handleToggleStatus(item);
   349→    }, show: canEdit, danger: false },
   350→    { label: "Hapus (Nonaktifkan)", icon: Trash2, action: (item: any) => {
   351→      setActionMenuId(null);
   352→      if (window.confirm(`Yakin nonaktifkan ${item.nama}? Data tidak akan dihapus permanen, hanya status diubah menjadi Nonaktif.`)) {
   353→        handleDelete(item);
   354→      }
   355→    }, show: canDelete, danger: true },
   356→  ];
   357→
   358→  return (
   359→    <div className="space-y-6">
   360→      {/* Header */}
   361→      <div className="flex items-center justify-between">
   362→        <div>
   363→          <h1 className="text-2xl font-bold text-blue-950">Master Wilayah</h1>
   364→          <p className="text-slate-500 text-sm mt-1">
   365→            Kelola data wilayah organisasi KIPAN: Nasional → Provinsi → Kabupaten/Kota
   366→          </p>
   367→        </div>
   368→      </div>
   369→
   370→      {/* Stat Cards */}
   371→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   372→        {statCards.map((s, idx) => {
   373→          const Icon = s.icon;
   374→          return (
   375→            <motion.div
   376→              key={idx}
   377→              initial={{ opacity: 0, y: 20 }}
   378→              animate={{ opacity: 1, y: 0 }}
   379→              transition={{ delay: idx * 0.08 }}
   380→              whileHover={{ y: -3 }}
   381→              onClick={() => s.targetPage && onNavigate?.(s.targetPage)}
   382→              className={`bg-white rounded-2xl p-5 shadow-sm border border-slate-100 ${s.targetPage ? "cursor-pointer hover:border-blue-200" : ""}`}
   383→            >
   384→              <div className={`w-11 h-11 rounded-xl bg-gradient-to-br ${s.color} flex items-center justify-center shadow-md mb-3`}>
   385→                <Icon className="w-6 h-6 text-white" />
   386→              </div>
   387→              <div className="text-3xl font-extrabold text-blue-950">
   388→                {s.value.toLocaleString("id-ID")}
   389→                {s.total && <span className="text-base text-slate-400">/{s.total}</span>}
   390→              </div>
   391→              <div className="text-sm text-slate-500 mt-1">{s.label}</div>
   392→            </motion.div>
   393→          );
   394→        })}
   395→      </div>
   396→
   397→      {/* Tabs */}
   398→      <div className="flex items-center justify-between">
   399→        <div className="flex gap-2 bg-white p-1 rounded-xl border border-slate-200">
   400→          <button
   401→            onClick={() => switchTab("provinsi")}
   402→            className={`px-4 py-2 rounded-lg text-sm font-medium transition-all ${
   403→              tab === "provinsi" ? "bg-blue-600 text-white" : "text-slate-600 hover:bg-slate-100"
   404→            }`}
   405→          >
   406→            <MapPin className="w-4 h-4 inline mr-1.5" />
   407→            Provinsi ({provData.length})
   408→          </button>
   409→          <button
   410→            onClick={() => switchTab("kabupaten")}
   411→            className={`px-4 py-2 rounded-lg text-sm font-medium transition-all ${
   412→              tab === "kabupaten" ? "bg-blue-600 text-white" : "text-slate-600 hover:bg-slate-100"
   413→            }`}
   414→          >
   415→            <Building2 className="w-4 h-4 inline mr-1.5" />
   416→            Kabupaten/Kota ({kabData.length})
   417→          </button>
   418→        </div>
   419→        {/* Tambah Provinsi disabled — all 38 provinsi already registered */}
   420→        {tab === "provinsi" && (
   421→          <div className="inline-flex items-center gap-2 px-4 py-2 bg-slate-100 text-slate-500 text-sm font-medium rounded-lg border border-slate-200">
   422→            <CheckCircle2 className="w-4 h-4 text-emerald-500" />
   423→            Semua 38 provinsi sudah terdaftar
   424→          </div>
   425→        )}
   426→        {tab === "kabupaten" && canCreateKabupaten && (
   427→          <button
   428→            onClick={() => { setFormData({ type: "kabupaten", kode: "", nama: "", status: "Aktif", ketua: "" }); setShowFormDialog(true); }}
   429→            className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 text-white text-sm font-semibold rounded-lg hover:bg-blue-700 transition-colors"
   430→          >
   431→            <Plus className="w-4 h-4" />
   432→            Tambah Kabupaten/Kota
   433→          </button>
   434→        )}
   435→      </div>
   436→
   437→      {/* Toolbar */}
   438→      <div className="bg-white rounded-xl border border-slate-200 p-4 flex flex-wrap items-center gap-3">
   439→        <div className="relative flex-1 min-w-[200px]">
   440→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   441→          <input
   442→            type="text"
   443→            value={search}
   444→            onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   445→            placeholder={tab === "provinsi" ? "Cari nama, kode, atau ketua provinsi..." : "Cari nama atau kode kabupaten..."}
   446→            className="w-full pl-9 pr-4 py-2 text-sm bg-slate-50 border border-slate-200 rounded-lg focus:border-blue-500 focus:ring-2 focus:ring-blue-500/20 outline-none"
   447→          />
   448→        </div>
   449→        {tab === "kabupaten" && (
   450→          <select
   451→            value={provinsiFilter}
   452→            onChange={(e) => { setProvinsiFilter(e.target.value); setPage(1); }}
   453→            className="px-3 py-2 text-sm bg-white border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   454→          >
   455→            <option value="Semua">Semua Provinsi</option>
   456→            {[...new Set(kabData.map((k: any) => k.provinsiNama))].map((p) => (
   457→              <option key={p}>{p}</option>
   458→            ))}
   459→          </select>
   460→        )}
   461→        <select
   462→          value={statusFilter}
   463→          onChange={(e) => { setStatusFilter(e.target.value); setPage(1); }}
   464→          className="px-3 py-2 text-sm bg-white border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   465→        >
   466→          <option value="Semua">Semua Status</option>
   467→          <option value="Aktif">Aktif</option>
   468→          <option value="Pembentukan">Pembentukan</option>
   469→          <option value="Nonaktif">Nonaktif</option>
   470→          <option value="Dibekukan">Dibekukan</option>
   471→        </select>
   472→        <div className="flex items-center gap-1">
   473→          <button
   474→            onClick={handleRefresh}

   517→      {/* Filter Summary */}

   545→          <div className="p-6 space-y-4">
   546→            {[1, 2, 3, 4, 5].map((i) => (

   548→                <div className="w-12 h-4 bg-slate-200 animate-pulse rounded" />
   549→                <div className="flex-1 h-4 bg-slate-200 animate-pulse rounded" />
   550→                <div className="w-20 h-4 bg-slate-200 animate-pulse rounded" />
   551→                <div className="w-16 h-4 bg-slate-200 animate-pulse rounded" />
   552→                <div className="w-24 h-4 bg-slate-200 animate-pulse rounded" />
   553→                <div className="w-20 h-4 bg-slate-200 animate-pulse rounded" />

   557→          </div>

   559→          /* Empty State */
   560→          <div className="p-16 text-center">
   561→            <div className="w-20 h-20 rounded-2xl bg-slate-100 flex items-center justify-center mx-auto mb-4">

   568→                : `Belum ada ${tab === "provinsi" ? "provinsi" : "kabupaten/kota"} yang terdaftar.`}

   573→                className="inline-flex items-center gap-2 px-4 py-2 bg-blue-600 text-white text-sm font-semibold rounded-lg hover:bg-blue-700"

   575→                <Plus className="w-4 h-4" />
   576→                Tambah Kabupaten/Kota
   577→              </button>
   578→            )}
   579→          </div>
   580→        ) : (
   581→          <div className="overflow-x-auto">
   582→            <table className="w-full">
   583→              <thead className="bg-slate-50 border-b border-slate-200 sticky top-0 z-10">
   584→                {tab === "provinsi" ? (
   585→                  <tr>
   586→                    <Th onClick={() => handleSort("kode")} icon={getSortIcon("kode")}>Kode</Th>
   587→                    <Th onClick={() => handleSort("nama")} icon={getSortIcon("nama")}>Nama Provinsi</Th>
   588→                    <th className="px-4 py-3 text-center text-xs font-bold text-slate-600 uppercase tracking-wider">Jml. Kab/Kota</th>
   589→                    <th className="px-4 py-3 text-center text-xs font-bold text-slate-600 uppercase tracking-wider">Jml. Pengurus</th>

   616→                      <td className="px-4 py-3 text-center">

   618→                          onClick={(e) => {
   619→                            e.stopPropagation();
   620→                            switchTab("kabupaten");
   621→                            setProvinsiFilter(item.nama);
   622→                            setPage(1);
   623→                          }}

   636→                        onClick={(e) => {
   637→                          e.stopPropagation();

   751→              >
   752→                <ChevronsUpDown className="w-4 h-4 -rotate-90" />

   773→            }

   782→          onNavigate?.("pengurus");
   783→        }}
   784→        onViewAnggota={(id) => {
   785→          setDetailId(null);
   786→          onNavigate?.("pengurus");
   787→        }}
   788→        onAddAnggota={() => {

   796→      />
   797→
   798→      {/* Form Dialog for Add/Edit */}
   799→      <WilayahFormDialog

   807→
   808→      {/* Backdrop for action menu */}
   809→      {actionMenuId !== null && (
   810→        <div
   811→          className="fixed inset-0 z-[5]"
   812→          onClick={() => setActionMenuId(null)}
   813→        />
   814→      )}
   815→    </div>
   816→  );
   817→}
