    87→  // Initial load + apply filter from navigation

    90→      if (initialFilter.provinsiNama) setProvinsiFilter(initialFilter.provinsiNama);

    99→  const canDelete = userRole === "SUPER_ADMIN";
   100→  const canEdit = ["SUPER_ADMIN", "ADMIN_NASIONAL", "ADMIN_PROVINSI"].includes(userRole);

   109→    if (lower === "provinsi") return "Provinsi";
   110→    if (lower === "kabupaten") return "Kabupaten";

   112→    return level;

   118→    foto: p.anggota?.foto || "https://images.unsplash.com/photo-1507003211169-0a1dd7228f2d?auto=format&fit=crop&w=400&q=80",

   124→    provinsiNama: p.provinsi?.nama || p.anggota?.provinsi?.nama,

   133→    anggotaId: p.anggotaId,

   145→  const pengurusNasional = pengurusData.filter((p) => p.level === "Nasional").length;

   149→    if (!p.tanggalSelesai) return false;

   152→    const diff = (end.getTime() - now.getTime()) / (1000 * 60 * 60 * 24);

   159→    { label: "Pengurus Provinsi", value: pengurusProvinsi, icon: Building2, color: "from-sky-500 to-cyan-500" },
   160→    { label: "Pengurus Kabupaten", value: pengurusKabupaten, icon: Building2, color: "from-cyan-500 to-teal-500" },
   161→    { label: "Masa Jabatan Akan Berakhir", value: masaJabatanBerakhir, icon: CalendarClock, color: "from-amber-500 to-orange-500" },

   175→        p.bidang.toLowerCase().includes(search.toLowerCase()) ||
   176→        p.nomorSK.toLowerCase().includes(search.toLowerCase());

   178→      const matchBidang = bidangFilter === "Semua" || p.bidang === bidangFilter;
   179→      const matchProv = provinsiFilter === "Semua" || p.provinsiNama === provinsiFilter;
   180→      const matchKab = kabupatenFilter === "Semua" || p.kabupatenNama === kabupatenFilter;

   183→      if (masaJabatanFilter === "Aktif") {
   184→        matchMasa = p.status === "Aktif" && (!p.tanggalSelesai || new Date(p.tanggalSelesai) > new Date());

   186→        if (!p.tanggalSelesai) matchMasa = false;

   188→          const diff = (new Date(p.tanggalSelesai).getTime() - new Date().getTime()) / (1000 * 60 * 60 * 24);

   192→        matchMasa = p.tanggalSelesai ? new Date(p.tanggalSelesai) < new Date() : false;
   193→      }

   201→          return sortDir === "asc" ? aVal.localeCompare(bVal) : bVal.localeCompare(aVal);
   202→        }
   203→        return sortDir === "asc" ? aVal - bVal : bVal - aVal;
   204→      });
   205→    }
   206→    return result;

   210→  const totalPages = Math.ceil(totalData / rowsPerPage) || 1;
   211→  const currentPage = Math.min(page, totalPages);
   212→  const startIdx = (currentPage - 1) * rowsPerPage;
   213→  const endIdx = Math.min(startIdx + rowsPerPage, totalData);
   214→  const pageData = filtered.slice(startIdx, endIdx);
   215→

   217→  const filterSummary = {
   218→    total: filtered.length,
   219→    nasional: filtered.filter((p) => p.level === "Nasional").length,
   220→    provinsi: filtered.filter((p) => p.level === "Provinsi").length,
   221→    kabupaten: filtered.filter((p) => p.level === "Kabupaten").length,
   222→    aktif: filtered.filter((p) => p.status === "Aktif").length,
   223→    nonaktif: filtered.filter((p) => p.status !== "Aktif").length,
   224→    akanBerakhir: filtered.filter((p) => {
   225→      if (!p.tanggalSelesai) return false;
   226→      const diff = (new Date(p.tanggalSelesai).getTime() - new Date().getTime()) / (1000 * 60 * 60 * 24);
   227→      return diff > 0 && diff < 365;
   228→    }).length,
   229→  };

   233→    if (sortBy === col) {
   234→      if (sortDir === "asc") { setSortDir("desc"); }
   235→      else if (sortDir === "desc") { setSortBy(null); setSortDir(null); }
   236→    } else {
   237→      setSortBy(col);
   238→      setSortDir("asc");
   239→    }
   240→  };
   241→
   242→  const getSortIcon = (col: string) => {
   243→    if (sortBy !== col) return <ArrowUpDown className="w-3 h-3 text-slate-300" />;
   244→    if (sortDir === "asc") return <ChevronUp className="w-3 h-3 text-blue-600" />;
   245→    if (sortDir === "desc") return <ChevronDown className="w-3 h-3 text-blue-600" />;
   246→    return <ArrowUpDown className="w-3 h-3 text-slate-300" />;
   247→  };
   248→
   249→  const handleRefresh = () => {
   250→    setLoading(true);
   251→    setSearch("");
   252→    setLevelFilter("Semua");
   253→    setProvinsiFilter("Semua");
   254→    setKabupatenFilter("Semua");
   255→    setStatusFilter("Semua");
   256→    setMasaJabatanFilter("Semua");
   257→    setSortBy(null);
   258→    setSortDir(null);
   259→    setPage(1);
   260→    setTimeout(() => setLoading(false), 600);
   261→  };
   262→
   263→  const statusBadge = (status: string) => {
   264→    const styles: Record<string, string> = {
   265→      Aktif: "bg-emerald-100 text-emerald-700 border-emerald-200",
   266→      Nonaktif: "bg-slate-100 text-slate-600 border-slate-200",
   267→      Dibekukan: "bg-rose-100 text-rose-700 border-rose-200",
   268→    };
   269→    return styles[status] || "bg-slate-100 text-slate-600 border-slate-200";
   270→  };
   271→
   272→  const levelBadge = (level: string) => {
   273→    const styles: Record<string, string> = {
   274→      Nasional: "bg-violet-100 text-violet-700",
   275→      Provinsi: "bg-blue-100 text-blue-700",
   276→      Kabupaten: "bg-cyan-100 text-cyan-700",
   277→    };
   278→    return styles[level] || "bg-slate-100 text-slate-600";
   279→  };
   280→

   282→    const m = new Date(mulai).toLocaleDateString("id-ID", { month: "short", year: "numeric" });
   283→    if (!selesai) return `${m} - Sekarang`;
   284→    const s = new Date(selesai).toLocaleDateString("id-ID", { month: "short", year: "numeric" });
   285→    return `${m} - ${s}`;
   286→  };

   288→  const handleSavePengurus = async (data: any) => {
   289→    const res = await fetch("/api/pengurus", {
   290→      method: "POST",
   291→      headers: { "Content-Type": "application/json" },
   292→      body: JSON.stringify(data),
   293→    });
   294→    const json = await res.json();
   295→    if (!json.success) throw new Error(json.error);
   296→    toast.success(json.message);
   297→    fetchData();
   298→  };
   299→
   300→  const actions = [
   301→    { label: "Detail", icon: Eye, action: (item: any) => setDetailId(item.id), show: true },
   302→    { label: "Edit", icon: Edit, action: (item: any) => { setActionMenuId(null); toast.info("Form edit pengurus akan dibuka"); }, show: canEdit },
   303→    { label: "Lihat Anggota", icon: Users, action: () => { setActionMenuId(null); onNavigate?.("anggota"); }, show: true },
   304→    { label: "Reset Password", icon: KeyRound, action: (item: any) => {
   305→      setActionMenuId(null);
   306→      toast.success(`Link reset password dikirim ke ${item.email}`);
   307→    }, show: canResetPassword },
   308→    { label: "Nonaktifkan", icon: Ban, action: (item: any) => {
   309→      setActionMenuId(null);
   310→      toast.success(`Pengurus ${item.nama} dinonaktifkan`);
   311→      fetchData();
   312→    }, show: canDelete, danger: false },
   313→    { label: "Hapus", icon: Trash2, action: (item: any) => {
   314→      setActionMenuId(null);
   315→      if (window.confirm(`Yakin hapus pengurus "${item.nama}"?`)) {
   316→        toast.success(`Pengurus ${item.nama} dihapus`);

   318→      }
   319→    }, show: canDelete, danger: true },

   324→      {/* Header */}

   345→        {statCards.map((s, idx) => {
   346→          const Icon = s.icon;
   347→          return (
   348→            <motion.div
   349→              key={idx}
   350→              initial={{ opacity: 0, y: 20 }}
   351→              animate={{ opacity: 1, y: 0 }}
   352→              transition={{ delay: idx * 0.08 }}
   353→              whileHover={{ y: -3 }}
   354→              className="bg-white rounded-2xl p-5 shadow-sm border border-slate-100"
   355→            >
   356→              <div className={`w-11 h-11 rounded-xl bg-gradient-to-br ${s.color} flex items-center justify-center shadow-md mb-3`}>
   357→                <Icon className="w-6 h-6 text-white" />
   358→              </div>
   359→              <div className="text-3xl font-extrabold text-blue-950">{s.value}</div>
   360→              <div className="text-xs text-slate-500 mt-1">{s.label}</div>
   361→            </motion.div>
   362→          );
   363→        })}
   364→      </div>
   365→
   366→      {/* Level Tabs — quick filter by level */}
   367→      <div className="flex gap-2 bg-white p-1 rounded-xl border border-slate-200 overflow-x-auto">
   368→        <button
   369→          onClick={() => { setLevelFilter("Semua"); setPage(1); }}
   370→          className={`px-4 py-2 rounded-lg text-sm font-medium transition-all whitespace-nowrap ${
   371→            levelFilter === "Semua" ? "bg-blue-600 text-white" : "text-slate-600 hover:bg-slate-100"
   372→          }`}
   373→        >
   374→          <Users className="w-4 h-4 inline mr-1.5" />
   375→          Semua Pengurus ({pengurusData.length})
   376→        </button>
   377→        <button

   380→            levelFilter === "Nasional" ? "bg-violet-600 text-white" : "text-slate-600 hover:bg-slate-100"
   381→          }`}
   382→        >
   383→          <Landmark className="w-4 h-4 inline mr-1.5" />

   398→            levelFilter === "Kabupaten" ? "bg-cyan-600 text-white" : "text-slate-600 hover:bg-slate-100"

   408→        <div className="relative flex-1 min-w-[200px]">

   412→            value={search}
   413→            onChange={(e) => { setSearch(e.target.value); setPage(1); }}
   414→            placeholder="Cari nama, jabatan, atau nomor SK..."
   415→            className="w-full pl-9 pr-4 py-2 text-sm bg-slate-50 border border-slate-200 rounded-lg focus:border-blue-500 focus:ring-2 focus:ring-blue-500/20 outline-none"
   416→          />
   417→        </div>

   420→          onChange={(e) => { setLevelFilter(e.target.value); setPage(1); }}

   422→        >
   423→          <option value="Semua">Semua Level</option>

   425→          <option value="Provinsi">Provinsi</option>

   427→        </select>

   429→          value={bidangFilter}

   432→        >
   433→          <option value="Semua">Semua Bidang</option>
   434→          {bidangOptions.map((b) => (
   435→            <option key={b} value={b}>{b}</option>
   436→          ))}
   437→        </select>
   438→        <select
   439→          value={provinsiFilter}
   440→          onChange={(e) => { setProvinsiFilter(e.target.value); setKabupatenFilter("Semua"); setPage(1); }}
   441→          className="px-3 py-2 text-sm bg-white border border-slate-200 rounded-lg focus:border-blue-500 outline-none"
   442→        >
   443→          <option value="Semua">Semua Provinsi</option>
   444→          {PROVINSI_LIST.map((p) => (
   445→            <option key={p.id}>{p.nama}</option>

   447→        </select>
