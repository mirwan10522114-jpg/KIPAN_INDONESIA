   560→                      rows={5}
   561→                      placeholder="Ceritakan motivasi Anda bergabung dengan KIPAN..."
   562→                    />
   563→                  </Field>
   564→
   565→                  {/* Upload dokumen placeholder */}
   566→                  <div>
   567→                    <label className="block text-xs font-semibold text-slate-700 mb-2">
   568→                      Upload Dokumen (KTP, Pas Foto, CV, Surat Pernyataan)
   569→                    </label>
   570→                    <div className="border-2 border-dashed border-sky-200 rounded-2xl p-6 text-center hover:border-sky-400 transition-colors cursor-pointer bg-sky-50/30">
   571→                      <Upload className="w-8 h-8 text-sky-400 mx-auto mb-2" />
   572→                      <p className="text-sm text-slate-600">
   573→                        Klik untuk upload atau drag &amp; drop file di sini
   574→                      </p>
   575→                      <p className="text-xs text-slate-400 mt-1">
   576→                        Format: JPG, PNG, PDF (max 2MB per file)
   577→                      </p>
   578→                      <input type="file" multiple className="hidden" />
   579→                    </div>
   580→                  </div>
   581→                </motion.div>
   582→              )}
   583→
   584→              {/* STEP 3: Persyaratan */}
   585→              {step === 3 && (
   586→                <motion.div
   587→                  key="step3"
   588→                  initial={{ opacity: 0, x: 20 }}
   589→                  animate={{ opacity: 1, x: 0 }}
   590→                  exit={{ opacity: 0, x: -20 }}
   591→                  className="space-y-4"
   592→                >
   593→                  <h3 className="text-xl font-bold text-blue-950 mb-2">
   594→                    Persyaratan Keanggotaan
   595→                  </h3>
   596→                  <p className="text-sm text-slate-600 mb-4">
   597→                    Centang seluruh persyaratan di bawah ini untuk melanjutkan.
   598→                  </p>
   599→
   600→                  <div className="space-y-3">
   601→                    {persyaratan.map((p, idx) => {
   602→                      const Icon = PERSYARATAN_ICONS[p.icon] || CheckCircle2;
   603→                      const checked = form.persyaratan[idx];
   604→                      return (
   605→                        <motion.label
   606→                          key={idx}
   607→                          initial={{ opacity: 0, y: 10 }}
   608→                          animate={{ opacity: 1, y: 0 }}
   609→                          transition={{ delay: idx * 0.05 }}
   610→                          className={`flex items-start gap-3 p-4 rounded-2xl border-2 cursor-pointer transition-all ${
   611→                            checked
   612→                              ? "border-sky-400 bg-sky-50"
   613→                              : "border-slate-200 hover:border-sky-300"
   614→                          }`}
   615→                        >
   616→                          <button
   617→                            type="button"
   618→                            onClick={() => {
   619→                              const next = [...form.persyaratan];
   620→                              next[idx] = !next[idx];
   621→                              update("persyaratan", next);
   622→                            }}
   623→                            className={`shrink-0 w-6 h-6 rounded-md border-2 flex items-center justify-center transition-all ${
   624→                              checked
   625→                                ? "bg-sky-500 border-sky-500"
   626→                                : "border-slate-300"
   627→                            }`}
   628→                          >
   629→                            {checked && <CheckCircle2 className="w-4 h-4 text-white" />}
   630→                          </button>
   631→                          <div className="w-10 h-10 rounded-xl bg-sky-100 flex items-center justify-center shrink-0">
   632→                            <Icon className="w-5 h-5 text-blue-600" />
   633→                          </div>
   634→                          <div className="flex-1">
   635→                            <div className="font-semibold text-slate-800 text-sm">
   636→                              {p.title}
   637→                            </div>
   638→                            <div className="text-xs text-slate-500 mt-0.5">
   639→                              {p.desc}
   640→                            </div>
   641→                          </div>
   642→                        </motion.label>
   643→                      );
   644→                    })}
   645→                  </div>
   646→
   647→                  {!allPersyaratanChecked && (
   648→                    <p className="text-xs text-amber-600 bg-amber-50 border border-amber-200 rounded-lg p-3">
   649→                      ⚠️ Semua persyaratan harus dicentang untuk melanjutkan ke tahap konfirmasi.
   650→                    </p>
   651→                  )}
   652→                </motion.div>
   653→              )}
   654→
   655→              {/* STEP 4: Konfirmasi */}
   656→              {step === 4 && (
   657→                <motion.div
   658→                  key="step4"
   659→                  initial={{ opacity: 0, x: 20 }}
   660→                  animate={{ opacity: 1, x: 0 }}
   661→                  exit={{ opacity: 0, x: -20 }}
   662→                  className="space-y-4"
   663→                >
   664→                  <h3 className="text-xl font-bold text-blue-950 mb-4">
   665→                    Konfirmasi Data
   666→                  </h3>
   667→
   668→                  <div className="bg-sky-50 border border-sky-200 rounded-2xl p-5 space-y-3">
   669→                    <div className="text-xs font-bold text-blue-700 uppercase tracking-wider">
   670→                      Data Diri
   671→                    </div>
   672→                    <Row label="Nama" value={form.namaLengkap} />
   673→                    <Row label="NIK" value={form.nik} />
   674→                    <Row label="TTL" value={`${form.tempatLahir}, ${form.tanggalLahir}`} />
   675→                    <Row label="Jenis Kelamin" value={form.jenisKelamin === "L" ? "Laki-laki" : form.jenisKelamin === "P" ? "Perempuan" : "-"} />
   676→                    <Row label="Alamat" value={`${form.alamat}, ${form.kecamatan}, ${form.kabupaten}, ${form.provinsi} ${form.kodePos}`} />
   677→                  </div>
   678→
   679→                  <div className="bg-sky-50 border border-sky-200 rounded-2xl p-5 space-y-3">
   680→                    <div className="text-xs font-bold text-blue-700 uppercase tracking-wider">
   681→                      Kontak
   682→                    </div>
   683→                    <Row label="Email" value={form.email} />
   684→                    <Row label="HP" value={form.nomorHP} />
   685→                    <Row label="WhatsApp" value={form.whatsapp} />
   686→                  </div>
   687→
   688→                  <div className="bg-amber-50 border border-amber-200 rounded-2xl p-4">
   689→                    <p className="text-xs text-amber-800 leading-relaxed">
   690→                      <strong>Pernyataan:</strong> Saya menyatakan bahwa data yang
   691→                      saya isi adalah benar dan dapat dipertanggungjawabkan. Saya
   692→                      bersedia mengikuti seluruh rangkaian verifikasi dan pelatihan
   693→                      yang diselenggarakan KIPAN.
   694→                    </p>
   695→                  </div>
   696→                </motion.div>
   697→              )}
   698→            </AnimatePresence>
   699→
   700→            {/* Navigation buttons */}
   701→            <div className="flex items-center justify-between mt-8 pt-6 border-t border-slate-100">
   702→              <button
   703→                onClick={prevStep}
   704→                disabled={step === 1}
   705→                className="inline-flex items-center gap-2 px-5 py-2.5 text-sm font-medium text-slate-600 hover:text-blue-700 disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
   706→              >
   707→                <ChevronLeft className="w-4 h-4" />
   708→                Kembali
   709→              </button>
   710→
   711→              {step < STEPS.length ? (
   712→                <button
   713→                  onClick={nextStep}
   714→                  disabled={(step === 3 && !allPersyaratanChecked)}
   715→                  className="inline-flex items-center gap-2 px-6 py-3 bg-gradient-to-r from-sky-500 to-blue-600 text-white font-semibold rounded-full shadow-lg shadow-sky-500/30 hover:shadow-xl hover:-translate-y-0.5 transition-all disabled:opacity-50 disabled:cursor-not-allowed disabled:hover:translate-y-0"
   716→                >
   717→                  Lanjut
   718→                  <ChevronRight className="w-4 h-4" />
   719→                </button>
   720→              ) : (
   721→                <div className="flex flex-col items-end gap-2">
   722→                  {submitError && (
   723→                    <div className="text-xs text-rose-600 bg-rose-50 border border-rose-200 rounded-lg px-3 py-2 max-w-xs">
   724→                      ⚠️ {submitError}
   725→                    </div>
   726→                  )}
   727→                  <button
   728→                    onClick={handleSubmit}
   729→                    disabled={submitting}
   730→                    className="inline-flex items-center gap-2 px-6 py-3 bg-gradient-to-r from-sky-500 to-blue-600 text-white font-semibold rounded-full shadow-lg shadow-sky-500/30 hover:shadow-xl hover:-translate-y-0.5 transition-all disabled:opacity-50 disabled:cursor-not-allowed disabled:hover:translate-y-0"
   731→                  >
   732→                    {submitting ? (
   733→                      <>
   734→                        <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   735→                        Mengirim...
   736→                      </>
   737→                    ) : (
   738→                      <>
   739→                        <Send className="w-4 h-4" />
   740→                        Kirim Pendaftaran
   741→                      </>
   742→                    )}
   743→                  </button>
   744→                </div>
   745→              )}
   746→            </div>
   747→          </motion.div>
   748→        </div>
   749→      </div>
   750→
   751→      {/* Inline style for form inputs */}
   752→      <style jsx global>{`
   753→        .form-input {
   754→          width: 100%;
   755→          padding: 0.625rem 0.875rem;
   756→          font-size: 0.875rem;
   757→          border-radius: 0.5rem;
   758→          border: 1px solid #e2e8f0;
   759→          background: white;
   760→          outline: none;
   761→          transition: all 0.2s;
   762→        }
   763→        .form-input:focus {
   764→          border-color: #10b981;
   765→          box-shadow: 0 0 0 3px rgba(16, 185, 129, 0.15);
   766→        }
   767→      `}</style>
   768→    </section>
   769→  );
   770→}
   771→
   772→function Field({
   773→  label,
   774→  required,
   775→  children,
   776→}: {
   777→  label: string;
   778→  required?: boolean;
   779→  children: React.ReactNode;
   780→}) {
   781→  return (
   782→    <label className="block">
   783→      <span className="block text-xs font-semibold text-slate-700 mb-1.5">
   784→        {label} {required && <span className="text-rose-500">*</span>}
   785→      </span>
   786→      {children}
   787→    </label>
   788→  );
   789→}
   790→
   791→function Row({ label, value }: { label: string; value: string }) {
   792→  return (
   793→    <div className="flex items-start gap-3 text-sm">
   794→      <span className="text-slate-500 min-w-[100px]">{label}</span>
   795→      <span className="text-slate-800 font-medium flex-1">{value || "-"}</span>
   796→    </div>
   797→  );
   798→}
